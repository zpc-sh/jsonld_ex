# BUILD: Simple Makefile for JsonldEx development
# No rustler_precompiled nonsense - just clean, reliable source builds

.PHONY: help clean dev prod test bench ci format lint docs
.PHONY: version-patch version-minor version-major
.PHONY: hex-build hex-publish hex-docs hex-auth
.PHONY: release-patch release-minor release-major

# Default target
.DEFAULT_GOAL := help

# BUILD: Colors for output
BLUE := \033[0;34m
GREEN := \033[0;32m
YELLOW := \033[1;33m
NC := \033[0m

# BUILD: Print help
help: ## Show this help message
	@echo "JsonldEx Simple Build System"
	@echo ""
	@echo "$(BLUE)Available targets:$(NC)"
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | sort | awk 'BEGIN {FS = ":.*?## "}; {printf "  $(GREEN)%-15s$(NC) %s\n", $$1, $$2}'
	@echo ""
	@echo "$(BLUE)Examples:$(NC)"
	@echo "  make dev         # Development build"
	@echo "  make prod        # Production build"
	@echo "  make test        # Run tests"
	@echo "  make release-patch # Release new patch version"

clean: ## Clean all build artifacts
	@echo "$(BLUE)[BUILD]$(NC) Cleaning all build artifacts..."
	mix clean --deps
	rm -rf _build
	rm -rf deps/_build
	rm -rf native/target

dev: ## Development build
	@echo "$(BLUE)[BUILD]$(NC) Development build..."
	mix deps.get
	mix compile

prod: ## Production build with optimizations
	@echo "$(BLUE)[BUILD]$(NC) Production build..."
	MIX_ENV=prod mix deps.get
	MIX_ENV=prod mix compile

test: ## Run tests
	@echo "$(BLUE)[TEST]$(NC) Running tests..."
	mix test

bench: ## Run benchmarks
	@echo "$(BLUE)[BENCH]$(NC) Running benchmarks..."
	mix run benchmark.exs

format: ## Format code
	@echo "$(BLUE)[FORMAT]$(NC) Formatting code..."
	mix format

lint: ## Run linter
	@echo "$(BLUE)[LINT]$(NC) Running linter..."
	mix credo --strict

docs: ## Generate documentation
	@echo "$(BLUE)[DOCS]$(NC) Generating documentation..."
	mix docs

ci: clean test lint docs ## Full CI pipeline
	@echo "$(GREEN)✓$(NC) CI pipeline completed!"

# Version management (simplified)
version-patch: ## Increment patch version
	@echo "$(BLUE)[VERSION]$(NC) Incrementing patch version..."
	@CURRENT_VERSION=$$(grep 'version:' mix.exs | sed 's/.*version: "\([^"]*\)".*/\1/'); \
	IFS='.' read -r major minor patch <<< "$$CURRENT_VERSION"; \
	NEW_PATCH=$$((patch + 1)); \
	NEW_VERSION="$$major.$$minor.$$NEW_PATCH"; \
	echo "$(GREEN)Updating version:$(NC) $$CURRENT_VERSION -> $$NEW_VERSION"; \
	sed -i.bak "s/version: \"$$CURRENT_VERSION\",/version: \"$$NEW_VERSION\",/" mix.exs && rm mix.exs.bak; \
	echo "$(GREEN)✓$(NC) Version updated to $$NEW_VERSION"

version-minor: ## Increment minor version
	@echo "$(BLUE)[VERSION]$(NC) Incrementing minor version..."
	@CURRENT_VERSION=$$(grep 'version:' mix.exs | sed 's/.*version: "\([^"]*\)".*/\1/'); \
	IFS='.' read -r major minor patch <<< "$$CURRENT_VERSION"; \
	NEW_MINOR=$$((minor + 1)); \
	NEW_VERSION="$$major.$$NEW_MINOR.0"; \
	echo "$(GREEN)Updating version:$(NC) $$CURRENT_VERSION -> $$NEW_VERSION"; \
	sed -i.bak "s/version: \"$$CURRENT_VERSION\",/version: \"$$NEW_VERSION\",/" mix.exs && rm mix.exs.bak; \
	echo "$(GREEN)✓$(NC) Version updated to $$NEW_VERSION"

version-major: ## Increment major version
	@echo "$(BLUE)[VERSION]$(NC) Incrementing major version..."
	@CURRENT_VERSION=$$(grep 'version:' mix.exs | sed 's/.*version: "\([^"]*\)".*/\1/'); \
	IFS='.' read -r major minor patch <<< "$$CURRENT_VERSION"; \
	NEW_MAJOR=$$((major + 1)); \
	NEW_VERSION="$$NEW_MAJOR.0.0"; \
	echo "$(GREEN)Updating version:$(NC) $$CURRENT_VERSION -> $$NEW_VERSION"; \
	sed -i.bak "s/version: \"$$CURRENT_VERSION\",/version: \"$$NEW_VERSION\",/" mix.exs && rm mix.exs.bak; \
	echo "$(GREEN)✓$(NC) Version updated to $$NEW_VERSION"

# Hex.pm publishing
hex-build: clean prod docs ## Build hex package
	@echo "$(BLUE)[HEX]$(NC) Building hex package..."
	mix hex.build

hex-publish: hex-build ## Publish to hex.pm
	@echo "$(BLUE)[HEX]$(NC) Publishing to hex.pm..."
	@CURRENT_VERSION=$$(grep 'version:' mix.exs | sed 's/.*version: "\([^"]*\)".*/\1/'); \
	echo "$(YELLOW)[CONFIRM]$(NC) About to publish version $$CURRENT_VERSION to hex.pm. Continue? [y/N]"; \
	read -r CONFIRM </dev/tty; \
	if [ "$$CONFIRM" = "y" ] || [ "$$CONFIRM" = "Y" ]; then \
		mix hex.publish; \
		echo "$(GREEN)✓$(NC) Published version $$CURRENT_VERSION to hex.pm!"; \
	else \
		echo "$(YELLOW)[CANCELLED]$(NC) Publish cancelled"; \
	fi

hex-docs: docs ## Publish documentation to hex.pm
	@echo "$(BLUE)[HEX]$(NC) Publishing documentation..."
	mix hex.publish docs

hex-auth: ## Authenticate with hex.pm
	@echo "$(BLUE)[HEX]$(NC) Authenticating with hex.pm..."
	mix hex.user auth

# Release workflows
release-patch: version-patch hex-publish hex-docs ## Complete patch release
	@echo "$(GREEN)✓$(NC) Patch release completed!"

release-minor: version-minor hex-publish hex-docs ## Complete minor release
	@echo "$(GREEN)✓$(NC) Minor release completed!"

release-major: version-major hex-publish hex-docs ## Complete major release
	@echo "$(GREEN)✓$(NC) Major release completed!"
EOF </dev/null
